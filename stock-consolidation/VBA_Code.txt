'==============================================================================
' STOCK CONSOLIDATION PLANNER
'
' HOW TO USE THIS CODE IN YOUR OWN EXCEL FILE
'   1. Open the Excel file with your report (headers in row 1: Prtnum, Stoloc,
'      Max of Curqvl, Max of Fp Available, Max of Maxqvl, Typcod, ...).
'   2. Press Alt+F11, then click Insert > Module.
'   3. Paste all of this code into the empty white window.
'   4. Close that window, press Alt+F8, choose BuildConsolidationPlan, click Run.
'   The sheets "Clean Data", "Consolidation Plan" and "Not Consolidated" are
'   created for you. To keep the macro, save the file as "Excel Macro-Enabled
'   Workbook (*.xlsm)". The settings are the DEFAULT_ lines below.
'
' POWER BI
'   If the report is a table exported from Power BI "with live connection",
'   the macro first refreshes it (like Data > Refresh All), so every run uses
'   the latest Power BI data.
'
' Reads the location report (a sheet called "Raw Data", or else the sheet that
' has the report headers), keeps only Prtnum, Stoloc,
' Curqvl, Fp Available, Maxqvl and Typcod, cleans it into "Clean Data", then
' finds items stored in more than one location and works out how many units
' to move from which location to which, so that locations can be emptied.
' It uses exactly the same rules as consolidation_plan.sql, so for the same
' data both give the same moves.
'
' Cleaning:
'   - rows with 0 or a negative current qty, max qty or available capacity
'     are removed (available capacity only while that setting is Yes);
'   - only locations whose max qty is from 9 to 23 are kept;
'   - the same item and location listed twice is kept once (largest values).
'
' Planning:
'   1. Only items sitting in 2 or more locations are looked at.
'   2. Locations are emptied smallest quantity first.
'   3. Stock only goes to locations that already hold the same item.
'   4. A location is only planned if it can be emptied completely:
'        - one move if any location has room for all of it (the one holding
'          the most of the item wins, then the one with least spare room);
'        - otherwise, if splitting is allowed, it is spread over the
'          locations with the most room first.
'   5. A location that is emptied never receives stock, and a location that
'      receives stock is never emptied.
'   6. Open capacity starts at Fp Available and goes down as moves are
'      planned (and up when a location is emptied), shared by all items in
'      the location.
'
' Report columns are found by their headers ("Prtnum", "Stoloc", "Max of
' Curqvl", ...) in any of the first 20 rows, in any order; other columns are
' ignored.
'==============================================================================
Option Explicit
Option Compare Binary

' Settings used when the file has no Start sheet with settings (for example
' when you paste this code into your own file). Change them here.
Private Const DEFAULT_MIN_LOCATION_QTY As Double = 9        ' only locations with max qty from this ...
Private Const DEFAULT_MAX_LOCATION_QTY As Double = 23       ' ... up to this
Private Const DEFAULT_REMOVE_NO_CAPACITY As String = "Yes"  ' remove rows with 0 / negative available capacity
Private Const DEFAULT_SAME_TYPE_ONLY As String = "No"       ' only move within the same location type
Private Const DEFAULT_ALLOW_SPLIT As String = "Yes"         ' allow emptying a location into several
Private Const DEFAULT_EXCLUDE_TYPES As String = ""          ' location types to ignore, e.g. "CONS"
Private Const DEFAULT_REFRESH_FIRST As String = "Yes"       ' refresh connected data (e.g. Power BI) first

Private Const EPS As Double = 0.000001
Private Const HEADER_ROW As Long = 4
Private Const FIRST_ROW As Long = 5
Private Const CLEAN_COLS As Long = 6
Private Const PLAN_COLS As Long = 9
Private Const FAIL_COLS As Long = 6

' Accepted headers (lower case, spaces as _, without a "Max of"/"Sum of" prefix)
Private Const ITEM_HEADERS As String = "prtnum,item_number,item,item_no,part_number,part_no,part,sku"
Private Const LOC_HEADERS As String = "stoloc,location,location_id,loc,bin"
Private Const QTY_HEADERS As String = "curqvl,current_qty,current_quantity,quantity,qty"
Private Const AVAIL_HEADERS As String = "fp_available,available_capacity,available,open_capacity"
Private Const MAX_HEADERS As String = "maxqvl,max_qty,max_quantity,max_capacity,capacity"
Private Const TYPE_HEADERS As String = "typcod,location_type,loc_type,type"

Private Const REASON_TYPE As String = "No other location of this item with the same location type"
Private Const REASON_ROOM As String = "Not enough room in the item's other locations"
Private Const REASON_SINGLE As String = "No single location of this item has room for all of it"

Private mMaxLocationQty As Double
Private mRemoveNoCapacity As Boolean

Private mExcludeText As String

Private mShowMessages As Boolean
Private mRefreshFirst As Boolean
Private mRefreshNote As String          ' what happened when refreshing connected data

Private statType As Long

Private rowCount As Long
Private rItem() As String
Private rItemKey() As String
Private rLoc() As String
Private rLocKey() As String

Private rAvail() As Double

Private rLocIdx() As Long

' Locations (1 To locCount); capacity is shared by every item in a location

Private lMax() As Double
Private lType() As String
Private lTypeKey() As String

'------------------------------------------------------------------------------
' Buttons
'------------------------------------------------------------------------------
Public Sub BuildConsolidationPlan()
    RunConsolidation True
End Sub

Public Sub ClearResults()
    Dim ws As Worksheet
    Set mBook = ActiveWorkbook
    Set ws = FindSheet(mBook, "shtClean", "Clean Data")
    If Not ws Is Nothing Then
        StartSheet ws, "Clean Data", "Press 'Build Consolidation Plan' on the Start sheet."
        WriteHeaders ws, CleanHeaders()
    End If
    Set ws = FindSheet(mBook, "shtPlan", "Consolidation Plan")
    If Not ws Is Nothing Then
        StartSheet ws, "Consolidation Plan", "Press 'Build Consolidation Plan' on the Start sheet."
        WriteHeaders ws, PlanHeaders()
    End If
    Set ws = FindSheet(mBook, "shtNotMoved", "Not Consolidated")
    If Not ws Is Nothing Then
        StartSheet ws, "Not Consolidated", "Press 'Build Consolidation Plan' on the Start sheet."
        WriteHeaders ws, FailHeaders()
    End If
    SetStatus "Results cleared " & Format$(Now, "dd-mmm-yyyy hh:mm")
End Sub

' Runs the whole thing. showMessages = False skips the message boxes.
Public Sub RunConsolidation(ByVal showMessages As Boolean)
    Dim oldCalc As Long, raw As Worksheet
    mShowMessages = showMessages
    oldCalc = Application.Calculation
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    On Error GoTo Failed

    Set mBook = ActiveWorkbook
    Set raw = FindRawSheet()
    If raw Is Nothing Then
        Report "Could not find the report. Paste it into a sheet with these headers in row 1: " & _
               "Prtnum, Stoloc, Max of Curqvl, Max of Fp Available, Max of Maxqvl, Typcod."
        GoTo Finish
    End If
    Set mBook = raw.Parent

    LoadSettings
    mRefreshNote = RefreshConnections()
    If LoadRawData(raw) Then
        PlanMoves
        WriteCleanData
        WritePlan
        WriteNotMoved
        SetStatus "Last run " & Format$(Now, "dd-mmm-yyyy hh:mm") & ": " & rowCount & " rows kept, " & _
                  moveCount & " moves, " & EmptiedCount() & " locations emptied, " & _
                  RemainingFailures() & " cannot be emptied" & StatusNote()
        OutputSheet("shtPlan", "Consolidation Plan").Activate
        If mShowMessages Then
            MsgBox "Plan ready." & vbCrLf & vbCrLf & MessageNote() & _
                   statRead & " report rows read, " & rowCount & " kept after cleaning." & vbCrLf & _
                   moveCount & " moves will empty " & EmptiedCount() & " locations (" & _
                   FormatQty(TotalMoved()) & " units)." & vbCrLf & _
                   RemainingFailures() & " locations cannot be emptied - see 'Not Consolidated'.", _
                   vbInformation, "Stock consolidation"
        End If
    End If

Finish:
    Application.Calculation = oldCalc
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    Exit Sub

Failed:
    SetStatus "Last run failed: error " & Err.Number & " - " & Err.Description
    If mShowMessages Then MsgBox "Error " & Err.Number & ": " & Err.Description, vbExclamation, "Stock consolidation"
    Resume Finish
End Sub

'------------------------------------------------------------------------------
' Input and cleaning
'------------------------------------------------------------------------------
Private Sub LoadSettings()
    Dim part As Variant, t As String
    mMinLocationQty = SettingNumber("MinLocationQty", DEFAULT_MIN_LOCATION_QTY)
    mMaxLocationQty = SettingNumber("MaxLocationQty", DEFAULT_MAX_LOCATION_QTY)
    mRemoveNoCapacity = IsYes(SettingText("RemoveNoCapacity", DEFAULT_REMOVE_NO_CAPACITY))
    mSameTypeOnly = IsYes(SettingText("SameTypeOnly", DEFAULT_SAME_TYPE_ONLY))
    mAllowSplit = IsYes(SettingText("AllowSplit", DEFAULT_ALLOW_SPLIT))
    mExcludeText = SettingText("ExcludeTypes", DEFAULT_EXCLUDE_TYPES)
    mRefreshFirst = IsYes(SettingText("RefreshFirst", DEFAULT_REFRESH_FIRST))
    Set mExcluded = New Collection
    For Each part In Split(mExcludeText, ",")
        t = UCase$(Trim$(CStr(part)))
        If Len(t) > 0 Then
            If Lookup(mExcluded, t) = 0 Then mExcluded.Add 1, t
        End If
    Next part
End Sub

Private Function LoadRawData(ByVal ws As Worksheet) As Boolean
    Dim data As Variant, rowIndex As Collection, locIndex As Collection
    Dim hdr As Long, lastCol As Long, lastRow As Long, n As Long, r As Long, idx As Long, li As Long
    Dim dropped As Long
    Dim cItem As Long, cLoc As Long, cQty As Long, cAvail As Long, cMax As Long, cType As Long
    Dim item As String, loc As String, typ As String, key As String
    Dim qty As Double, avail As Double, mx As Double
    Dim hasQty As Boolean, hasAvail As Boolean, hasMax As Boolean

    hdr = FindHeaderRow(ws)
    If hdr = 0 Then hdr = 1
    lastCol = FindLastColumn(ws, hdr)
    cItem = FindColumn(ws, hdr, lastCol, ITEM_HEADERS)
    cLoc = FindColumn(ws, hdr, lastCol, LOC_HEADERS)
    cQty = FindColumn(ws, hdr, lastCol, QTY_HEADERS)
    cAvail = FindColumn(ws, hdr, lastCol, AVAIL_HEADERS)
    cMax = FindColumn(ws, hdr, lastCol, MAX_HEADERS)
    cType = FindColumn(ws, hdr, lastCol, TYPE_HEADERS)
    If cItem = 0 Or cLoc = 0 Or cQty = 0 Or cMax = 0 Then
        Report "The '" & ws.Name & "' sheet needs a header row with: Prtnum, Stoloc, Max of Curqvl, " & _
               "Max of Maxqvl (and ideally Max of Fp Available and Typcod)."
        Exit Function
    End If

    lastRow = FindLastRow(ws, cItem, cLoc, cQty)
    n = lastRow - hdr
    If n < 1 Then n = 1
    ReDim rItem(1 To n)
    ReDim rItemKey(1 To n)
    ReDim rLoc(1 To n)
    ReDim rLocKey(1 To n)
    ReDim rQty(1 To n)
    ReDim rAvail(1 To n)
    ReDim rMax(1 To n)
    ReDim rType(1 To n)
    Set rowIndex = New Collection
    rowCount = 0
    statRead = 0
    statZero = 0
    statOutOfRange = 0
    statType = 0

    If lastRow > hdr Then
        data = ws.Range(ws.Cells(hdr + 1, 1), ws.Cells(lastRow, lastCol)).Value
        For r = 1 To UBound(data, 1)
            item = CellText(data(r, cItem))
            loc = CellText(data(r, cLoc))
            If Len(item) > 0 And Len(loc) > 0 Then
                statRead = statRead + 1
                typ = ""
                If cType > 0 Then typ = CellText(data(r, cType))
                hasQty = IsNumber(data(r, cQty))
                hasMax = IsNumber(data(r, cMax))
                hasAvail = False
                If cAvail > 0 Then hasAvail = IsNumber(data(r, cAvail))
                If hasQty Then qty = CDbl(data(r, cQty))
                If hasMax Then mx = CDbl(data(r, cMax))
                If hasAvail Then
                    avail = CDbl(data(r, cAvail))
                ElseIf hasQty And hasMax Then
                    avail = mx - qty                ' no Fp Available: max - current
                    hasAvail = True
                End If

                dropped = 0
                If Not hasQty Or Not hasMax Then
                    dropped = 1
                ElseIf qty <= 0 Or mx <= 0 Then
                    dropped = 1
                ElseIf mRemoveNoCapacity And Not hasAvail Then
                    dropped = 1
                ElseIf mRemoveNoCapacity And avail <= 0 Then
                    dropped = 1
                ElseIf mx > mMaxLocationQty Or mx < mMinLocationQty Then
                    dropped = 2
                ElseIf Lookup(mExcluded, UCase$(typ)) > 0 Then
                    dropped = 3
                End If
                If Not hasAvail Then avail = 0

                Select Case dropped
                Case 1
                    statZero = statZero + 1
                Case 2
                    statOutOfRange = statOutOfRange + 1
                Case 3
                    statType = statType + 1
                Case Else
                    key = UCase$(item) & vbTab & UCase$(loc)
                    idx = Lookup(rowIndex, key)
                    If idx = 0 Then
                        rowCount = rowCount + 1
                        idx = rowCount
                        rowIndex.Add idx, key
                        rItem(idx) = item
                        rItemKey(idx) = UCase$(item)
                        rLoc(idx) = loc
                        rLocKey(idx) = UCase$(loc)
                        rQty(idx) = qty
                        rAvail(idx) = avail
                        rMax(idx) = mx
                        rType(idx) = typ
                    Else                            ' listed twice: keep the largest values
                        If item < rItem(idx) Then rItem(idx) = item
                        If loc < rLoc(idx) Then rLoc(idx) = loc
                        If qty > rQty(idx) Then rQty(idx) = qty
                        If avail > rAvail(idx) Then rAvail(idx) = avail
                        If mx > rMax(idx) Then rMax(idx) = mx
                        If typ > rType(idx) Then rType(idx) = typ
                    End If
                End Select
            End If
        Next r
    End If

    SortByItemAndLocation

    ' Location level values
    ReDim rCur(1 To n)
    ReDim rLocIdx(1 To n)
    ReDim rState(1 To n)
    ReDim rMulti(1 To n)
    ReDim lOpen(1 To n)
    ReDim lMax(1 To n)
    ReDim lType(1 To n)
    ReDim lTypeKey(1 To n)
    Set locIndex = New Collection
    locCount = 0
    For idx = 1 To rowCount
        rCur(idx) = rQty(idx)
        li = Lookup(locIndex, rLocKey(idx))
        If li = 0 Then
            locCount = locCount + 1
            li = locCount
            locIndex.Add li, rLocKey(idx)
            lOpen(li) = rAvail(idx)
            lMax(li) = rMax(idx)
            lType(li) = rType(idx)
        Else
            If rAvail(idx) > lOpen(li) Then lOpen(li) = rAvail(idx)
            If rMax(idx) > lMax(li) Then lMax(li) = rMax(idx)
            If rType(idx) > lType(li) Then lType(li) = rType(idx)
        End If
        rLocIdx(idx) = li
    Next idx
    For li = 1 To locCount
        lTypeKey(li) = UCase$(lType(li))
    Next li
    LoadRawData = True
End Function

' Puts the cleaned rows in item, then location order
Private Sub SortByItemAndLocation()
    Dim ord() As Long, i As Long
    Dim a1() As String, a2() As String, a3() As String, a4() As String, a5() As String
    Dim b1() As Double, b2() As Double, b3() As Double
    If rowCount < 2 Then Exit Sub
    ReDim ord(1 To rowCount)
    For i = 1 To rowCount
        ord(i) = i
    Next i
    SortIndex ord, 1, rowCount, 1
    ReDim a1(1 To rowCount)
    ReDim a2(1 To rowCount)
    ReDim a3(1 To rowCount)
    ReDim a4(1 To rowCount)
    ReDim a5(1 To rowCount)
    ReDim b1(1 To rowCount)
    ReDim b2(1 To rowCount)
    ReDim b3(1 To rowCount)
    For i = 1 To rowCount
        a1(i) = rItem(ord(i))
        a2(i) = rItemKey(ord(i))
        a3(i) = rLoc(ord(i))
        a4(i) = rLocKey(ord(i))
        a5(i) = rType(ord(i))
        b1(i) = rQty(ord(i))
        b2(i) = rAvail(ord(i))
        b3(i) = rMax(ord(i))
    Next i
    rItem = a1
    rItemKey = a2
    rLoc = a3
    rLocKey = a4
    rType = a5
    rQty = b1
    rAvail = b2
    rMax = b3
End Sub

'------------------------------------------------------------------------------
' Planning
'------------------------------------------------------------------------------
Private Sub PlanMoves()
    Dim ord() As Long, i As Long, k As Long, g1 As Long, g2 As Long

    moveCount = 0
    failCount = 0
    itemsMulti = 0
    ReDim moveList(1 To 2 * rowCount + 1, 1 To PLAN_COLS)
    ReDim failRow(1 To rowCount + 1)
    ReDim failRoom(1 To rowCount + 1)
    ReDim failReason(1 To rowCount + 1)
    If rowCount = 0 Then Exit Sub

    ReDim ord(1 To rowCount)
    For i = 1 To rowCount
        ord(i) = i
    Next i
    SortIndex ord, 1, rowCount, 2

    ' Walk through the items; within an item, smallest quantity first
    g1 = 1
    Do While g1 <= rowCount
        g2 = g1
        Do While g2 < rowCount
            If rItemKey(ord(g2 + 1)) <> rItemKey(ord(g1)) Then Exit Do
            g2 = g2 + 1
        Loop
        If g2 > g1 Then
            itemsMulti = itemsMulti + 1
            For k = g1 To g2
                rMulti(ord(k)) = True
            Next k
            For i = g1 To g2
                If rState(ord(i)) = 0 Then TryToEmpty ord, g1, g2, ord(i)
            Next i
        End If
        g1 = g2 + 1
    Loop
End Sub

' Tries to empty row s into the other locations of the same item (ord(g1..g2)).
Private Sub TryToEmpty(ord() As Long, ByVal g1 As Long, ByVal g2 As Long, ByVal s As Long)
    Dim cand() As Long, nc As Long, nOther As Long
    Dim i As Long, k As Long, t As Long, best As Long
    Dim need As Double, room As Double, sumOpen As Double, maxOpen As Double
    Dim remaining As Double, take As Double

    need = rQty(s)
    ReDim cand(1 To g2 - g1 + 1)
    For i = g1 To g2
        t = ord(i)
        If t <> s And rState(t) <> 1 Then
            If SameTypeOk(s, t) Then
                nOther = nOther + 1
                room = OpenCap(t)
                If room > EPS Then
                    nc = nc + 1
                    cand(nc) = t
                    sumOpen = sumOpen + room
                    If room > maxOpen Then maxOpen = room
                    If room >= need - EPS Then
                        If best = 0 Then
                            best = t
                        ElseIf SingleBefore(t, best) Then
                            best = t
                        End If
                    End If
                End If
            End If
        End If
    Next i

    If best > 0 Then
        AddMove s, best, need
    ElseIf mAllowSplit And nc > 0 And sumOpen >= need - EPS Then
        SortCandidates cand, nc
        remaining = need
        For k = 1 To nc
            If remaining <= EPS Then Exit For
            take = OpenCap(cand(k))
            If take > remaining Then take = remaining
            AddMove s, cand(k), take
            remaining = remaining - take
        Next k
    Else
        failCount = failCount + 1
        failRow(failCount) = s
        If mAllowSplit Then failRoom(failCount) = sumOpen Else failRoom(failCount) = maxOpen
        If nOther = 0 Then
            failReason(failCount) = REASON_TYPE
        ElseIf mAllowSplit Then
            failReason(failCount) = REASON_ROOM
        Else
            failReason(failCount) = REASON_SINGLE
        End If
        Exit Sub
    End If

    ' The source location is now empty, so its space is free again
    rState(s) = 1
    rCur(s) = 0
    lOpen(rLocIdx(s)) = lOpen(rLocIdx(s)) + need
End Sub

Private Sub AddMove(ByVal s As Long, ByVal t As Long, ByVal qty As Double)
    Dim li As Long, before As Double
    li = rLocIdx(t)
    before = lOpen(li)
    moveCount = moveCount + 1
    moveList(moveCount, 1) = rItem(s)
    moveList(moveCount, 2) = rLoc(s)
    moveList(moveCount, 3) = qty
    moveList(moveCount, 4) = rLoc(t)
    moveList(moveCount, 5) = before
    moveList(moveCount, 6) = lMax(li)
    moveList(moveCount, 7) = lType(li)
    moveList(moveCount, 8) = lType(rLocIdx(s))
    moveList(moveCount, 9) = before - qty
    lOpen(li) = before - qty
    rCur(t) = rCur(t) + qty
    rState(t) = 2
End Sub

Private Function SameTypeOk(ByVal s As Long, ByVal t As Long) As Boolean
    If mSameTypeOnly Then
        SameTypeOk = (lTypeKey(rLocIdx(s)) = lTypeKey(rLocIdx(t)))
    Else
        SameTypeOk = True
    End If
End Function

Private Function OpenCap(ByVal t As Long) As Double
    OpenCap = lOpen(rLocIdx(t))
End Function

' Sort orders: 1 = item, location (clean data)   2 = item, smallest qty, location (order of work)
Private Function RowBefore(ByVal a As Long, ByVal b As Long, ByVal order As Long) As Boolean
    If rItemKey(a) <> rItemKey(b) Then
        RowBefore = (rItemKey(a) < rItemKey(b))
    ElseIf order = 2 And Abs(rQty(a) - rQty(b)) > EPS Then
        RowBefore = (rQty(a) < rQty(b))
    Else
        RowBefore = (rLocKey(a) < rLocKey(b))
    End If
End Function

' Single target: holds the most of the item, then least spare room, then location
Private Function SingleBefore(ByVal a As Long, ByVal b As Long) As Boolean
    If Abs(rCur(a) - rCur(b)) > EPS Then
        SingleBefore = (rCur(a) > rCur(b))
    ElseIf Abs(OpenCap(a) - OpenCap(b)) > EPS Then
        SingleBefore = (OpenCap(a) < OpenCap(b))
    Else
        SingleBefore = (rLocKey(a) < rLocKey(b))
    End If
End Function

' Split targets: most spare room, then holds the most of the item, then location
Private Function SplitBefore(ByVal a As Long, ByVal b As Long) As Boolean
    If Abs(OpenCap(a) - OpenCap(b)) > EPS Then
        SplitBefore = (OpenCap(a) > OpenCap(b))
    ElseIf Abs(rCur(a) - rCur(b)) > EPS Then
        SplitBefore = (rCur(a) > rCur(b))
    Else
        SplitBefore = (rLocKey(a) < rLocKey(b))
    End If
End Function

Private Sub SortIndex(arr() As Long, ByVal lo As Long, ByVal hi As Long, ByVal order As Long)
    Dim i As Long, j As Long, p As Long, tmp As Long
    Do While lo < hi
        i = lo
        j = hi
        p = arr((lo + hi) \ 2)
        Do While i <= j
            Do While RowBefore(arr(i), p, order)
                i = i + 1
            Loop
            Do While RowBefore(p, arr(j), order)
                j = j - 1
            Loop
            If i <= j Then
                tmp = arr(i)
                arr(i) = arr(j)
                arr(j) = tmp
                i = i + 1
                j = j - 1
            End If
        Loop
        If j - lo < hi - i Then
            If lo < j Then SortIndex arr, lo, j, order
            lo = i
        Else
            If i < hi Then SortIndex arr, i, hi, order
            hi = j
        End If
    Loop
End Sub

' Few candidates per item, so a simple insertion sort is enough
Private Sub SortCandidates(arr() As Long, ByVal n As Long)
    Dim i As Long, j As Long, v As Long
    For i = 2 To n
        v = arr(i)
        j = i - 1
        Do While j >= 1
            If Not SplitBefore(v, arr(j)) Then Exit Do
            arr(j + 1) = arr(j)
            j = j - 1
        Loop
        arr(j + 1) = v
    Next i
End Sub

Private Sub WriteCleanData()
    Dim ws As Worksheet, out() As Variant, r As Long, lastRow As Long, fmt As String

    Set ws = OutputSheet("shtClean", "Clean Data")
    StartSheet ws, "Clean Data", "Rows read: " & statRead & _
               "   |   Removed - zero or negative qty/capacity: " & statZero & _
               "   |   Removed - max qty outside " & FormatQty(mMinLocationQty) & "-" & _
               FormatQty(mMaxLocationQty) & ": " & statOutOfRange & _
               "   |   Removed - ignored type: " & statType & _
               "   |   Kept: " & rowCount
    WriteHeaders ws, CleanHeaders()

    If rowCount > 0 Then
        ReDim out(1 To rowCount, 1 To CLEAN_COLS)
        fmt = "#,##0"
        For r = 1 To rowCount
            out(r, 1) = rItem(r)
            out(r, 2) = rLoc(r)
            out(r, 3) = rQty(r)
            out(r, 4) = rAvail(r)
            out(r, 5) = rMax(r)
            out(r, 6) = rType(r)
            If NotWhole(rQty(r)) Or NotWhole(rAvail(r)) Or NotWhole(rMax(r)) Then fmt = "#,##0.00"
        Next r
        lastRow = FIRST_ROW + rowCount - 1
        TextColumns ws, lastRow, Array(1, 2, 6)
        ws.Range(ws.Cells(FIRST_ROW, 1), ws.Cells(lastRow, CLEAN_COLS)).Value = out
        With ws.Range(ws.Cells(FIRST_ROW, 1), ws.Cells(lastRow, CLEAN_COLS))
            .Font.Size = 10
            .VerticalAlignment = xlCenter
            .HorizontalAlignment = xlCenter
            .Borders.LineStyle = xlContinuous
            .Borders.Color = RGB(217, 217, 217)
        End With
        ws.Range(ws.Cells(FIRST_ROW, 1), ws.Cells(lastRow, 1)).HorizontalAlignment = xlLeft
        ws.Range(ws.Cells(FIRST_ROW, 3), ws.Cells(lastRow, 5)).NumberFormat = fmt
        ' Highlight the items that sit in 2 or more locations
        For r = 1 To rowCount
            If rMulti(r) Then
                ws.Range(ws.Cells(FIRST_ROW + r - 1, 1), ws.Cells(FIRST_ROW + r - 1, CLEAN_COLS)).Interior.Color = RGB(255, 242, 204)
                ws.Cells(FIRST_ROW + r - 1, 1).Font.Bold = True
            End If
        Next r
    Else
        NoRowsMessage ws, "No rows left after cleaning - check the Raw Data sheet and the settings."
    End If

    ws.Range("H4").Value = "Item in 2+ locations"
    With ws.Range("H4")
        .Interior.Color = RGB(255, 242, 204)
        .Font.Bold = True
        .HorizontalAlignment = xlCenter
        .VerticalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(217, 217, 217)
    End With

    SetWidths ws, Array(14, 12, 12, 13, 10, 13, 3, 22)
    FinishSheet ws, CLEAN_COLS, rowCount
End Sub

Private Sub WritePlan()
    Dim ws As Worksheet, out() As Variant, r As Long, c As Long, lastRow As Long

    Set ws = OutputSheet("shtPlan", "Consolidation Plan")
    StartSheet ws, "Consolidation Plan", SettingsLine()
    WriteHeaders ws, PlanHeaders()

    If moveCount > 0 Then
        ReDim out(1 To moveCount, 1 To PLAN_COLS)
        For r = 1 To moveCount
            For c = 1 To PLAN_COLS
                out(r, c) = moveList(r, c)
            Next c
        Next r
        lastRow = FIRST_ROW + moveCount - 1
        TextColumns ws, lastRow, Array(1, 2, 4, 7, 8)
        ws.Range(ws.Cells(FIRST_ROW, 1), ws.Cells(lastRow, PLAN_COLS)).Value = out
        FormatTable ws, out, moveCount, PLAN_COLS, Array(3, 5, 6, 9), Array(1)
        With ws.Range(ws.Cells(FIRST_ROW, 3), ws.Cells(lastRow, 3))
            .Font.Bold = True
            .Font.Color = RGB(0, 97, 0)
        End With
    Else
        NoRowsMessage ws, "No moves found - nothing can be consolidated with this data and these settings."
    End If

    ' Summary panel
    ws.Range("K4").Value = "Summary"
    With ws.Range("K4:L4")
        .VerticalAlignment = xlCenter
        .Font.Bold = True
        .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(31, 56, 100)
    End With
    SummaryLine ws, 5, "Items in 2+ locations", itemsMulti
    SummaryLine ws, 6, "Locations emptied", EmptiedCount()
    SummaryLine ws, 7, "Moves", moveCount
    SummaryLine ws, 8, "Units to move", TotalMoved()
    SummaryLine ws, 9, "Locations that stay", RemainingFailures()
    ws.Range("L8").NumberFormat = QtyFormat(TotalMoved())
    With ws.Range("K4:L9").Borders
        .LineStyle = xlContinuous
        .Color = RGB(200, 206, 216)
    End With

    SetWidths ws, Array(14, 15, 12, 14, 13, 13, 13, 13, 14, 3, 22, 10)
    FinishSheet ws, PLAN_COLS, moveCount
End Sub

Private Sub WriteNotMoved()
    Dim ws As Worksheet, out() As Variant, idx() As Long, n As Long, i As Long, s As Long, lastRow As Long

    Set ws = OutputSheet("shtNotMoved", "Not Consolidated")
    StartSheet ws, "Not Consolidated", _
               "Locations of multi-location items that cannot be emptied with the space available.   " & SettingsLine()
    WriteHeaders ws, FailHeaders()

    ' Only locations that are still untouched at the end of planning
    If failCount > 0 Then ReDim idx(1 To failCount)
    For i = 1 To failCount
        If rState(failRow(i)) = 0 Then
            n = n + 1
            idx(n) = i
        End If
    Next i

    If n > 0 Then
        ReDim out(1 To n, 1 To FAIL_COLS)
        For i = 1 To n
            s = failRow(idx(i))
            out(i, 1) = rItem(s)
            out(i, 2) = rLoc(s)
            out(i, 3) = lType(rLocIdx(s))
            out(i, 4) = rQty(s)
            out(i, 5) = failRoom(idx(i))
            out(i, 6) = failReason(idx(i))
        Next i
        lastRow = FIRST_ROW + n - 1
        TextColumns ws, lastRow, Array(1, 2, 3, 6)
        ws.Range(ws.Cells(FIRST_ROW, 1), ws.Cells(lastRow, FAIL_COLS)).Value = out
        FormatTable ws, out, n, FAIL_COLS, Array(4, 5), Array(1, 6)
    Else
        NoRowsMessage ws, "Every multi-location item can be consolidated."
    End If

    SetWidths ws, Array(14, 14, 13, 12, 14, 56)
    FinishSheet ws, FAIL_COLS, n
End Sub

        .Value = title
        .Font.Size = 16
        .Font.Bold = True
        .Font.Color = RGB(31, 56, 100)
    End With
    With ws.Range("A2")
        .Value = subtitle
        .Font.Size = 9
        .Font.Color = RGB(89, 89, 89)
    End With
End Sub

    Next c
End Sub

' Borders, number formats and a light band for every other item
Private Sub FormatTable(ByVal ws As Worksheet, out() As Variant, ByVal n As Long, ByVal nCols As Long, _
                        ByVal qtyCols As Variant, ByVal leftCols As Variant)
    Dim lastRow As Long, r As Long, g1 As Long, shade As Boolean, c As Variant, fmt As String

    lastRow = FIRST_ROW + n - 1
    With ws.Range(ws.Cells(FIRST_ROW, 1), ws.Cells(lastRow, nCols))
        .Font.Size = 10
        .VerticalAlignment = xlCenter
        .HorizontalAlignment = xlCenter
        .Borders.LineStyle = xlContinuous
        .Borders.Color = RGB(217, 217, 217)
    End With
    For Each c In leftCols
        ws.Range(ws.Cells(FIRST_ROW, c), ws.Cells(lastRow, c)).HorizontalAlignment = xlLeft
    Next c

    fmt = "#,##0"
    For Each c In qtyCols
        For r = 1 To n
            If NotWhole(out(r, c)) Then fmt = "#,##0.00"
        Next r
    Next c
    For Each c In qtyCols
        ws.Range(ws.Cells(FIRST_ROW, c), ws.Cells(lastRow, c)).NumberFormat = fmt
    Next c

    g1 = 1
    For r = 1 To n
        If r = n Then
            BandGroup ws, FIRST_ROW + g1 - 1, FIRST_ROW + r - 1, nCols, shade
        ElseIf UCase$(out(r + 1, 1)) <> UCase$(out(r, 1)) Then
            BandGroup ws, FIRST_ROW + g1 - 1, FIRST_ROW + r - 1, nCols, shade
            shade = Not shade
            g1 = r + 1
        End If
    Next r
End Sub

Private Sub BandGroup(ByVal ws As Worksheet, ByVal r1 As Long, ByVal r2 As Long, ByVal nCols As Long, ByVal shade As Boolean)
    With ws.Range(ws.Cells(r1, 1), ws.Cells(r2, nCols))
        If shade Then .Interior.Color = RGB(234, 241, 251)
        With .Borders(xlEdgeTop)
            .LineStyle = xlContinuous
            .Color = RGB(142, 160, 189)
        End With
    End With
    ws.Cells(r1, 1).Font.Bold = True
End Sub

Private Sub FinishSheet(ByVal ws As Worksheet, ByVal nCols As Long, ByVal n As Long)

        .FreezePanes = False
        .ScrollRow = 1
        .ScrollColumn = 1
        .SplitColumn = 0
        .SplitRow = HEADER_ROW
        .FreezePanes = True

End Sub

                   "   |   Split moves: " & YesNo(mAllowSplit) & _
                   "   |   Ignored types: " & ex
End Function

' Finds a sheet by its code name, or failing that by its tab name
Private Function FindSheet(ByVal wb As Workbook, ByVal codeName As String, ByVal tabName As String) As Worksheet
    Dim ws As Worksheet
    For Each ws In wb.Worksheets
        If ws.CodeName = codeName Then
            Set FindSheet = ws
            Exit Function
        End If
    Next ws
    For Each ws In wb.Worksheets
        If StrComp(ws.Name, tabName, vbTextCompare) = 0 Then
            Set FindSheet = ws
            Exit Function
        End If
    Next ws
End Function

' A results sheet; it is added at the end if the workbook does not have it yet
Private Function OutputSheet(ByVal codeName As String, ByVal tabName As String) As Worksheet
    Dim ws As Worksheet
    Set ws = FindSheet(mBook, codeName, tabName)
    If ws Is Nothing Then
        Set ws = mBook.Worksheets.Add(After:=mBook.Worksheets(mBook.Worksheets.Count))
        ws.Name = tabName
    End If
    Set OutputSheet = ws
End Function

' The report: a sheet called "Raw Data", otherwise the active sheet or the
' first sheet whose row 1 has the report headers
Private Function FindRawSheet() As Worksheet
    Dim ws As Worksheet
    Set ws = FindSheet(mBook, "shtRaw", "Raw Data")
    If ws Is Nothing Then Set ws = FindSheet(ThisWorkbook, "shtRaw", "Raw Data")
    If Not ws Is Nothing Then
        Set FindRawSheet = ws
        Exit Function
    End If
    If TypeName(ActiveSheet) = "Worksheet" Then
        If IsReportSheet(ActiveSheet) Then
            Set FindRawSheet = ActiveSheet
            Exit Function
        End If
    End If
    For Each ws In mBook.Worksheets
        If IsReportSheet(ws) Then
            Set FindRawSheet = ws
            Exit Function
        End If
    Next ws
End Function

Private Function IsReportSheet(ByVal ws As Worksheet) As Boolean
    Select Case LCase$(ws.Name)
    Case "start", "clean data", "consolidation plan", "not consolidated"
        Exit Function                   ' never read our own sheets as the report
    End Select
    IsReportSheet = (FindHeaderRow(ws) > 0)
End Function

' Row (1 to 20) holding the report headers, or 0 when there is none
Private Function FindHeaderRow(ByVal ws As Worksheet) As Long
    Dim r As Long, lastCol As Long
    For r = 1 To 20
        lastCol = FindLastColumn(ws, r)
        If lastCol >= 4 Then
            If FindColumn(ws, r, lastCol, ITEM_HEADERS) > 0 Then
                If FindColumn(ws, r, lastCol, LOC_HEADERS) > 0 Then
                    If FindColumn(ws, r, lastCol, QTY_HEADERS) > 0 Then
                        If FindColumn(ws, r, lastCol, MAX_HEADERS) > 0 Then
                            FindHeaderRow = r
                            Exit Function
                        End If
                    End If
                End If
            End If
        End If
    Next r
End Function

' Pulls the latest data into every connected table of the workbook (for example
' a table exported from Power BI "with live connection") and waits until it is
' done. Returns "" when there was nothing to refresh, "OK" when it worked, or
' the names of the connections that could not be refreshed.
Private Function RefreshConnections() As String
    Dim cn As Object, n As Long, failed As String
    If Not mRefreshFirst Then Exit Function
    On Error Resume Next
    n = mBook.Connections.Count
    If Err.Number <> 0 Or n = 0 Then Exit Function
    Application.StatusBar = "Refreshing data from Power BI / connected sources..."
    For Each cn In mBook.Connections
        Err.Clear
        Select Case cn.Type
        Case 1                              ' OLE DB: Power BI, Power Query, SQL Server
            cn.OLEDBConnection.BackgroundQuery = False
        Case 2                              ' ODBC
            cn.ODBCConnection.BackgroundQuery = False
        End Select
        Err.Clear
        cn.Refresh                          ' waits, because background refresh is off
        If Err.Number <> 0 Then
            If Len(failed) > 0 Then failed = failed & ", "
            failed = failed & cn.Name
        End If
    Next cn
    Application.CalculateUntilAsyncQueriesDone
    Application.StatusBar = False
    Err.Clear
    On Error GoTo 0
    If Len(failed) = 0 Then RefreshConnections = "OK" Else RefreshConnections = failed
End Function

Private Function StatusNote() As String
    If mRefreshNote = "OK" Then
        StatusNote = " (data refreshed first)"
    ElseIf Len(mRefreshNote) > 0 Then
        StatusNote = " (could not refresh " & mRefreshNote & " - used the data already in the sheet)"
    End If
End Function

Private Function MessageNote() As String
    If mRefreshNote = "OK" Then
        MessageNote = "Data refreshed from its source (e.g. Power BI) first." & vbCrLf
    ElseIf Len(mRefreshNote) > 0 Then
        MessageNote = "Could not refresh " & mRefreshNote & ", so the data already in the sheet was used." & vbCrLf
    End If
End Function

' Column whose header (in row hdr) matches one of the comma-separated names
Private Function FindColumn(ByVal ws As Worksheet, ByVal hdr As Long, ByVal lastCol As Long, _
                            ByVal aliases As String) As Long
    Dim c As Long, h As String, nm As Variant, prefix As Variant
    For c = 1 To lastCol
        h = LCase$(CellText(ws.Cells(hdr, c).Value))
        For Each prefix In Array("max of ", "sum of ", "min of ", "average of ", "count of ")
            If Left$(h, Len(prefix)) = prefix Then h = Mid$(h, Len(prefix) + 1)
        Next prefix
        h = Replace(Replace(Trim$(h), " ", "_"), "-", "_")
        For Each nm In Split(aliases, ",")
            If h = nm Then
                FindColumn = c
                Exit Function
            End If
        Next nm
    Next c
End Function

Private Function FindLastColumn(ByVal ws As Worksheet, ByVal hdr As Long) As Long
    FindLastColumn = ws.Cells(hdr, ws.Columns.Count).End(xlToLeft).Column
End Function

Private Function FindLastRow(ByVal ws As Worksheet, ParamArray cols() As Variant) As Long
    Dim c As Variant, r As Long
    For Each c In cols
        r = ws.Cells(ws.Rows.Count, c).End(xlUp).Row
        If r > FindLastRow Then FindLastRow = r
    Next c
End Function

Private Function Lookup(ByVal col As Collection, ByVal key As String) As Long
    On Error Resume Next
    Lookup = col(key)
    On Error GoTo 0
End Function

Private Function CellText(ByVal v As Variant) As String
    If IsError(v) Or IsEmpty(v) Or IsNull(v) Then
        CellText = ""
    Else
        CellText = Trim$(CStr(v))
    End If
End Function

Private Function IsNumber(ByVal v As Variant) As Boolean
    If IsError(v) Or IsEmpty(v) Or IsNull(v) Then Exit Function
    If VarType(v) = vbBoolean Then Exit Function
    IsNumber = IsNumeric(v)
End Function

Private Function IsYes(ByVal s As String) As Boolean
    s = UCase$(Trim$(s))
    IsYes = (s = "YES" Or s = "Y" Or s = "TRUE" Or s = "1")
End Function

Private Function YesNo(ByVal b As Boolean) As String
    If b Then YesNo = "Yes" Else YesNo = "No"
End Function

Private Function NotWhole(ByVal v As Double) As Boolean
    NotWhole = (Abs(v - Fix(v)) > EPS)
End Function

Private Function QtyFormat(ByVal v As Double) As String
    If NotWhole(v) Then QtyFormat = "#,##0.00" Else QtyFormat = "#,##0"
End Function

Private Function FormatQty(ByVal v As Double) As String
    FormatQty = Format$(v, QtyFormat(v))
End Function
